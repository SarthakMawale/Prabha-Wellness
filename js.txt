                const selectedCategory = categoryFilter.value.toLowerCase();
                let hasVisibleProduct = false;

                    productCards.forEach(card => {
                        // Safety Check
                        const titleEl = card.querySelector('h3');
                        const badgeEl = card.querySelector('.badge');
                        const descEl = card.querySelector('.short-info');


                        const title = titleEl ? titleEl.innerText.toLowerCase() : "";
                        const badgeText = badgeEl ? badgeEl.innerText.toLowerCase() : "";

                        const matchesSearch = (searchTerm === '') ||
                            title.includes(searchTerm) ||
                            desc.includes(searchTerm);


                        // 2. CATEGORY MATCH
                        let matchesCategory = false;
                        if (selectedCategory === 'all') {
                            matchesCategory = true;

                            if (badgeText.includes(selectedCategory)) matchesCategory = true;
                            // Extra Mappings
                            if (selectedCategory === 'energy' && (badgeText.includes('sports') || title.includes('afresh'))) matchesCategory = true;
                            if (selectedCategory === 'joint' && (badgeText.includes('bone') || badgeText.includes('calcium'))) matchesCategory = true;
                            if (selectedCategory === 'digestive' && (badgeText.includes('gut') || badgeText.includes('aloe'))) matchesCategory = true;
                        }


                        // 3. DISPLAY RESULT
                        if (matchesSearch && matchesCategory) {
                            card.style.display = "block";

        // Toast Function (Also used by auth.js)
        function showToast(message, isError = false) {
            const toast = document.getElementById("toastBox");
            const msg = document.getElementById("toastMessage");
            const icon = toast.querySelector("i");
            msg.innerText = message;
            if (isError) {
                icon.className = "fas fa-exclamation-circle";
                icon.style.color = "#ff5252";
            } else {
                icon.className = "fas fa-check-circle";
                icon.style.color = "#4CAF50";
            }
            toast.classList.add("show");
            setTimeout(() => { toast.classList.remove("show"); }, 3000);
        }
